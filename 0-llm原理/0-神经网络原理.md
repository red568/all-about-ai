## 单向注意力机制和双向注意力机制

在模型阅读或生成文本时，注意力机制决定了它能“看到”哪些词汇来获取上下文：

- **双向注意力（Bidirectional）**：模型在处理一个词时，可以同时看到它**前面和后面**的所有词。这种方式具有“全局视野”，非常适合用来**理解**一段已经存在的文本。

- **单向注意力（Unidirectional / Causal）**：模型只能看到当前词以及它**前面**的词（严格从左到右），看不到未来的词。这种方式像人类说话一样，非常适合用来逐字逐句地**生成**文本。



## decode\-only架构

整体结构可以概括为三段式——输入层、N 层 Decoder Block 堆叠、输出层

```Python
B = 2           # batch size，一次处理的样本个数
T = 4           # 序列长度，每个样本有几个token
V = 10          # 词表大小,也就是这个模型可以识别多少token
d_model = 8     # 隐藏维度，每个 token 用多长的向量表示
n_heads = 2     # 注意力头数
d_head = 4      # 每头维度 = d_model / n_heads
n_layers = 2    # Decoder Block 层数
d_ff = 16       # FFN 中间维度
```

```Python
import math
import torch
import torch.nn as nn
import torch.nn.functional as F

torch.manual_seed(0)

class CausalSelfAttention(nn.Module):
    def __init__(self, d_model, n_heads, dropout=0.0):
        super().__init__()
        # 隐藏层维度必须可以整除头数
        assert d_model % n_heads == 0  
        self.d_model = d_model
        self.n_heads = n_heads
        self.d_head = d_model // n_heads

        # 一次线性映射得到 Q、K、V
        self.qkv = nn.Linear(d_model, 3 * d_model, bias=False)
        self.proj = nn.Linear(d_model, d_model, bias=False)
        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        B, T, C = x.shape
        # x: [B, T, C]

        qkv = self.qkv(x)
        # qkv: [B, T, 3C]

        q, k, v = qkv.chunk(3, dim=-1)
        # q, k, v: 每个都是 [B, T, C]

        # 拆多头
        q = q.reshape(B, T, self.n_heads, self.d_head).transpose(1, 2)
        k = k.reshape(B, T, self.n_heads, self.d_head).transpose(1, 2)
        v = v.reshape(B, T, self.n_heads, self.d_head).transpose(1, 2)
        # q, k, v: [B, n_heads, T, d_head]

        # 注意力分数
        scores = (q @ k.transpose(-2, -1)) / math.sqrt(self.d_head)
        # scores: [B, n_heads, T, T]

        # 因果 mask：位置 i 只能看 <= i
        mask = torch.triu(
            torch.ones(T, T, device=x.device, dtype=torch.bool),
            diagonal=1
        )
        # mask: [T, T]，上三角为 True
        scores = scores.masked_fill(mask, float("-inf"))

        attn = F.softmax(scores, dim=-1)
        # attn: [B, n_heads, T, T]
        attn = self.dropout(attn)

        y = attn @ v
        # y: [B, n_heads, T, d_head]

        # 合并多头
        y = y.transpose(1, 2).contiguous().view(B, T, C)
        # y: [B, T, C]

        return self.proj(y)
        # [B, T, C]


class MLP(nn.Module):
    def __init__(self, d_model, d_ff, dropout=0.0):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(d_model, d_ff),
            nn.GELU(),
            nn.Linear(d_ff, d_model),
            nn.Dropout(dropout),
        )

    def forward(self, x):
        return self.net(x)


class DecoderBlock(nn.Module):
    def __init__(self, d_model, n_heads, d_ff, dropout=0.0):
        super().__init__()
        self.ln1 = nn.LayerNorm(d_model)
        self.attn = CausalSelfAttention(d_model, n_heads, dropout)
        self.ln2 = nn.LayerNorm(d_model)
        self.mlp = MLP(d_model, d_ff, dropout)

    def forward(self, x):
        # Pre-Norm + 残差
        x = x + self.attn(self.ln1(x))
        x = x + self.mlp(self.ln2(x))
        return x


class DecoderOnlyLM(nn.Module):
    def __init__(self, vocab_size, d_model, n_heads, n_layers, d_ff, max_seq_len):
        super().__init__()
        self.token_emb = nn.Embedding(vocab_size, d_model)
        self.pos_emb = nn.Embedding(max_seq_len, d_model)

        self.blocks = nn.ModuleList([
            DecoderBlock(d_model, n_heads, d_ff)
            for _ in range(n_layers)
        ])

        self.ln_f = nn.LayerNorm(d_model)
        self.lm_head = nn.Linear(d_model, vocab_size, bias=False)

        # 常见做法：LM Head 与 token embedding 权重共享
        self.lm_head.weight = self.token_emb.weight

    def forward(self, idx):
        B, T = idx.shape
        # idx: [B, T]

        pos = torch.arange(T, device=idx.device)
        # pos: [T]

        x = self.token_emb(idx) + self.pos_emb(pos)[None, :, :]
        # token_emb(idx): [B, T, d_model]
        # pos_emb(pos):   [T, d_model]
        # x:              [B, T, d_model]

        for block in self.blocks:
            x = block(x)
            # x: [B, T, d_model]

        x = self.ln_f(x)
        # x: [B, T, d_model]

        logits = self.lm_head(x)
        # logits: [B, T, V]

        return logits
```
